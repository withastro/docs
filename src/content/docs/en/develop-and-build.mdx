      const campos = ['idioma', 'ano', 'titulo', 'autora', 'corrente', 'imagemCapa', 'arquivoAnexo', 'videoUrl', 'resumo', 'corpo'];
      const preview = document.getElementById('preview');

      function slugify(texto) {
        return (texto || 'novo-texto')
          .toLowerCase()
          .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
          .replace(/[^a-z0-9]+/g, '-')
          .replace(/(^-|-$)/g, '');
      }

      function gerarMarkdown() {
        const val = (id) => document.getElementById(id).value.trim();
        const linhas = [
          '---',
          `titulo: "${val('titulo').replace(/"/g, '\\"') || 'Sem título'}"`,
          `autora: "${val('autora').replace(/"/g, '\\"') || 'Autora Desconhecida'}"`,
          `idioma: "${val('idioma')}"`,
          val('ano') ? `ano: ${val('ano')}` : null,
          val('corrente') ? `corrente: "${val('corrente').replace(/"/g, '\\"')}"` : null,
          `tipo: "misto"`,
          val('imagemCapa') ? `imagemCapa: "${val('imagemCapa')}"` : null,
          val('arquivoAnexo') ? `arquivoAnexo: "${val('arquivoAnexo')}"` : null,
          val('videoUrl') ? `videoUrl: "${val('videoUrl')}"` : null,
          `resumo: "${val('resumo').replace(/"/g, '\\"') || 'Documento do acervo.'}"`,
          '---',
          '',
          val('corpo')
        ].filter((l) => l !== null);

        const md = linhas.join('\n');
        preview.textContent = md;

        const rascunho = {};
        campos.forEach((c) => rascunho[c] = document.getElementById(c).value);
        localStorage.setItem('rascunho_arquivo_feminista', JSON.stringify(rascunho));
        return md;
      }

      const salvo = localStorage.getItem('rascunho_arquivo_feminista');
      if (salvo) {

        campos.forEach((c) => { if (dados[c]) document.getElementById(c).value = dados[c]; });

      campos.forEach((c) => document.getElementById(c).addEventListener('input', gerarMarkdown));
